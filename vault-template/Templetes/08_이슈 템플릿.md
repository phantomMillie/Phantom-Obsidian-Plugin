<%*
/*
 * 이슈 등록 템플릿 (⌘⇧B)
 * 책 제목 / 화면명 / 컨텐츠 타입 / 이슈 항목 을 받아
 * 📥 이슈 인박스.md 의 `## 미분류` 바로 아래에 한 줄로 넣는다.
 * 임시 파일(issue_tmp)은 실행 후 삭제한다.
 */

const INBOX = "3. Resources/03. 개발일지/📥 이슈 인박스.md";
const HEADING = "## 미분류";

const cleanup = async (openInbox) => {
  tp.hooks.on_all_templates_executed(async () => {
    const tmpFile = tp.file.find_tfile(tp.file.path(true));
    if (tmpFile) {
      await tp.app.vault.trash(tmpFile, true);
    }
    if (openInbox) {
      const inbox = tp.app.vault.getAbstractFileByPath(INBOX);
      if (inbox) {
        await tp.app.workspace.getLeaf().openFile(inbox);
      }
    }
  });
};

// 1. 책 제목
const book = await tp.system.prompt("책 제목", "");
if (!book || !book.trim()) {
  new Notice("취소됨 — 책 제목 없음");
  await cleanup(false);
} else {
  // 2. 화면명
  const screen = await tp.system.prompt("화면명 (예: 설정, 홈)", "");

  // 3. 컨텐츠 타입
  const types = ["전자책", "오디오북", "챗북", "종이책", "PDF", "직접 입력"];
  let contentType = await tp.system.suggester(types, types, false, "컨텐츠 타입");
  if (contentType === "직접 입력") {
    contentType = await tp.system.prompt("컨텐츠 타입 직접 입력", "");
  }

  // 4. 이슈 항목
  const issue = await tp.system.prompt("이슈 항목 (증상 한 줄)", "");

  if (!issue || !issue.trim()) {
    new Notice("취소됨 — 이슈 항목 없음");
    await cleanup(false);
  } else {
    const timestamp = tp.date.now("YYYY-MM-DD HH:mm");
    const parts = [
      `[${book.trim()}]`,
      screen && screen.trim() ? screen.trim() : null,
      contentType && contentType.trim() ? contentType.trim() : null,
    ].filter(Boolean);
    const line = `- \`${timestamp}\` ${parts.join(" · ")} — ${issue.trim()}`;

    const inbox = tp.app.vault.getAbstractFileByPath(INBOX);
    if (!inbox) {
      new Notice(`이슈 인박스 없음: ${INBOX}`);
      await cleanup(false);
    } else {
      const content = await tp.app.vault.read(inbox);
      const idx = content.indexOf(HEADING);
      let updated;
      if (idx === -1) {
        // 헤딩이 없으면 파일 끝에 헤딩과 함께 추가
        updated = `${content.replace(/\s+$/, "")}\n\n${HEADING}\n\n${line}\n`;
      } else {
        const insertAt = idx + HEADING.length;
        const head = content.slice(0, insertAt);
        const tail = content.slice(insertAt).replace(/^\n+/, "");
        updated = `${head}\n\n${line}\n${tail ? `${tail}` : ""}`;
      }
      await tp.app.vault.modify(inbox, updated);
      new Notice(`이슈 등록됨 — ${book.trim()}`);
      await cleanup(false);
    }
  }
}
%>
