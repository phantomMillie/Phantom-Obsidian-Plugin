<%*
/*
 * Daily Development Log Template
 * Creates a new daily log on first call, appends logs on subsequent calls
 */

// 프로젝트 폴더 목록 동적 조회
const projectFolder = tp.app.vault.getAbstractFileByPath("1. Projects");
const projects = projectFolder.children
  .filter(f => f.children)
  .map(f => f.name)
  .filter(n => !n.startsWith("📓"));

const project = await tp.system.suggester(projects, projects, true, "프로젝트를 선택하세요");

// 파일명 설정
const fileName = tp.date.now("yyyy-MM-DD");
const devFolder = `3. Resources/03. 개발일지/${project} 개발일지`;
const filePath = `${devFolder}/${fileName}.md`;

// 폴더 생성
if (!tp.app.vault.getAbstractFileByPath(devFolder)) {
  await tp.app.vault.createFolder(devFolder);
}

// 파일 존재 여부 확인
const existingFile = tp.app.vault.getAbstractFileByPath(filePath);

if (existingFile) {
  // 파일이 있으면: 로그 입력받고 기존 파일에 추가
  const newLog = await tp.system.prompt("오늘의 로그를 입력하세요:", "");
  
  if (newLog && newLog.trim()) {
    const timestamp = tp.date.now("HH:mm");
    const fileContent = await tp.app.vault.read(existingFile);
    const trimmedContent = fileContent.replace(/\s+$/, "");
    const updatedContent = `${trimmedContent}\n- \`${timestamp}\` ${newLog}\n`;
    await tp.app.vault.modify(existingFile, updatedContent);
  }
  
  // 임시 파일(dev_tmp) 삭제 후 기존 파일 열기
  tp.hooks.on_all_templates_executed(async () => {
    const tmpFile = tp.file.find_tfile(tp.file.path(true));
    if (tmpFile) {
      await tp.app.vault.trash(tmpFile, true);
    }
    await tp.app.workspace.getLeaf().openFile(existingFile);
  });
} else {
  // 파일이 없으면: 새 파일 생성
  await tp.file.move(`${devFolder}/${fileName}`);

  const projectLink = `[[1. Projects/${project}/📓 ${project} Overview|🔗 ${project}]]`;

  tR += `---
title: Dev Log - ${fileName}
date: ${tp.date.now("yyyy-MM-DD")}
project: ${project}
tags: [Development, Daily]
---

${projectLink}

## ${fileName}

### 한 일
- 

### 결정
- 무엇을: 
- 왜 (대안 + 기각 사유): 

### 이슈
- 증상 / 원인 / 해결: 
- 수치: before → after

### 외부
- 

### 내일
- TODO: 
- 막힌 거 / 도움 필요: 

---

## 로그
- 
`;

  // 새 파일을 활성 leaf에 명시적으로 열기 (quickadd openFile=false 대응)
  tp.hooks.on_all_templates_executed(async () => {
    const newFile = tp.app.vault.getAbstractFileByPath(filePath);
    if (newFile) {
      await tp.app.workspace.getLeaf().openFile(newFile);
    }
  });
}
%>
